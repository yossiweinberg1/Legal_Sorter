import json
import os
from pathlib import Path

# 🔧 Set this to your actual pull folder
PULL_FOLDER = Path("Case_Pulls")   # or cfg["pull_folder"]

# 🔧 Load the bulk file name from your state file
STATE_FILE = Path("index/bulk_ingest_state.json")  # adjust if needed

if STATE_FILE.exists():
    state = json.loads(STATE_FILE.read_text(encoding="utf-8"))
    bulk_url = state.get("url")
    if bulk_url:
        bulk_file = bulk_url.split("/")[-1]
    else:
        bulk_file = "unknown-bulk-file.csv.bz2"
else:
    bulk_file = "unknown-bulk-file.csv.bz2"

print(f"[REPAIR] Using bulk file identifier: {bulk_file}")

count_fixed = 0
count_skipped = 0

for meta_path in PULL_FOLDER.glob("bulk_*.txt.meta.json"):
    try:
        data = json.loads(meta_path.read_text(encoding="utf-8"))

        # Extract opinion ID from filename
        opinion_id = meta_path.name.replace("bulk_", "").replace(".txt.meta.json", "")

        # Skip if already fixed
        if data.get("source_url") and data["source_url"] != "None":
            count_skipped += 1
            continue

        # Build synthetic URL
        synthetic_url = f"bulk://{bulk_file}#{opinion_id}"
        data["source_url"] = synthetic_url

        # Write back
        meta_path.write_text(json.dumps(data), encoding="utf-8")
        count_fixed += 1

    except Exception as e:
        print(f"[REPAIR] Failed to fix {meta_path}: {e}")

print(f"[REPAIR] Completed. Fixed {count_fixed} files, skipped {count_skipped}.")
